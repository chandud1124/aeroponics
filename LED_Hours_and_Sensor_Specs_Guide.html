Seeds do NOT need light to germinate. In fact, light can inhibit germination in some lettuce varieties. Keep seeds in a dark or very dim tray with a humidity dome. Light the tray only AFTER cotyledons (seed leaves) open and stand upright.
When cotyledons appear (Day 3–5): turn on LED for 4 hours/day at very low intensity — 500–1000 lux is enough. This is just to prevent etiolation (stretching toward darkness).

NEVER run lights 24 hours/day for lettuce. Lettuce requires a minimum 6-hour dark period daily for proper cellular respiration and nutrient uptake. 24-hour light causes tipburn, premature bolting, leaf edge whitening, and poor flavour. Always give 6 hours minimum darkness.
Timer: 4:00 AM ON → 10:00 PM OFF = 18 hrs light, 6 hrs dark
This is the maximum photoperiod for lettuce. Longer than this accelerates bolting.

3. Note ADC voltage reading — set as calibration point
4. Adjust K value in code to match known EC
5. Verify with second known solution (500 ppm or 2000 ppm)

Complete sensor reading code
// ─────────────────────────────────────────────────────
//  ALL SENSOR INCLUDES
// ─────────────────────────────────────────────────────
#include <Wire.h>
#include <BH1750.h>
#include <DHT.h>
#include <OneWire.h>
#include <DallasTemperature.h>
#include <GravityTDS.h>

// ─── GPIO PINS ─────────────────────────────────────
#define PIN_DS18B20   18   // Water temp probe
#define PIN_DHT22     19   // Humidity + air temp
#define PIN_EC_ANALOG 34   // EC/TDS sensor (ADC)
#define PIN_PH_ANALOG 35   // pH sensor (ADC)
#define PIN_PUMP_RELAY 27  // Pump relay
#define PIN_LED_RELAY  25  // LED light relay
// BH1750: GPIO21 (SDA), GPIO22 (SCL) — default I2C
// A02YYUW: GPIO16 (RX2), GPIO17 (TX2) — UART2

// ─── OBJECTS ───────────────────────────────────────
BH1750 lightMeter;
DHT dht(PIN_DHT22, DHT22);
OneWire oneWire(PIN_DS18B20);
DallasTemperature waterTempSensor(&oneWire);
GravityTDS gravityTds;

// ─── pH calibration constants (set after calibration) ──
const float PH_SLOPE     = -5.7;   // Adjust after 2-point cal
const float PH_INTERCEPT = 20.5;   // Adjust after 2-point cal

// ─── LED light control ──────────────────────────────
const int LED_ON_HOUR    = 5;    // 5:00 AM lights ON
const int LED_OFF_HOUR   = 23;   // 11:00 PM lights OFF (18hr photoperiod)
const int LED_PREHARVEST_OFF = 21; // 9:00 PM for pre-harvest (16hr)

// ─────────────────────────────────────────────────────
//  SETUP
// ─────────────────────────────────────────────────────
void setup() {
  Serial.begin(115200);

  // I2C for BH1750
  Wire.begin(21, 22);
  lightMeter.begin(BH1750::CONTINUOUS_HIGH_RES_MODE);

  // DHT22
  dht.begin();

  // DS18B20 water temp
  waterTempSensor.begin();
  waterTempSensor.setResolution(12); // 12-bit = 0.0625°C resolution

  // EC/TDS sensor
  gravityTds.setPin(PIN_EC_ANALOG);
  gravityTds.setAref(3.3);         // ESP32 ADC reference
  gravityTds.setAdcRange(4096);    // ESP32 = 12-bit = 4096
  gravityTds.begin();

  // UART2 for A02YYUW ultrasonic
  Serial2.begin(9600, SERIAL_8N1, 16, 17); // RX=16, TX=17

  // Relay pins
  pinMode(PIN_PUMP_RELAY, OUTPUT);
  pinMode(PIN_LED_RELAY,  OUTPUT);
  digitalWrite(PIN_PUMP_RELAY, HIGH); // HIGH = relay OFF (active LOW)
  digitalWrite(PIN_LED_RELAY,  HIGH);
}

// ─────────────────────────────────────────────────────
//  READ ALL SENSORS (call every 30 seconds)
// ─────────────────────────────────────────────────────
struct SensorReadings {
  float   lux;           // BH1750 — light intensity
  float   humidity;      // DHT22
  float   airTempC;      // DHT22
  float   waterTempC;    // DS18B20
  float   ecMS;          // EC sensor mS/cm
  float   ppm;           // TDS/PPM (EC × 700)
  float   pH;            // pH sensor
  int     waterLevelPct; // A02YYUW level %
  bool    valid;
};

const float TANK_HEIGHT_CM = 60.0; // Edit: your reservoir height in cm

SensorReadings readAllSensors() {
  SensorReadings r;
  r.valid = true;

  // 1. BH1750 — light (lux)
  r.lux = lightMeter.readLightLevel();
  Serial.printf("Light: %.0f lux\n", r.lux);

  // 2. DHT22 — humidity and air temperature
  r.humidity  = dht.readHumidity();
  r.airTempC  = dht.readTemperature();
  if (isnan(r.humidity) || isnan(r.airTempC)) {
    Serial.println("DHT22 read error!");
    r.valid = false;
  } else {
    Serial.printf("Air: %.1f°C  RH: %.1f%%\n", r.airTempC, r.humidity);
  }

  // 3. DS18B20 — water temperature (wait for 12-bit conversion)
  waterTempSensor.requestTemperatures();
  unsigned long convStart = millis();
  while (!waterTempSensor.isConversionComplete()) {
    if (millis() - convStart > 900) break; // 900ms timeout
    yield();
  }
  r.waterTempC = waterTempSensor.getTempCByIndex(0);
  if (r.waterTempC == DEVICE_DISCONNECTED_C) {
    Serial.println("DS18B20 disconnected!");
    r.valid = false;
  } else {
    Serial.printf("Water temp: %.2f°C\n", r.waterTempC);
  }

  // 4. EC/TDS sensor (with temperature compensation)
  gravityTds.setTemperature(r.waterTempC); // temp compensation
  gravityTds.update();
  r.ppm  = gravityTds.getTdsValue();      // PPM reading
  r.ecMS = r.ppm / 700.0;               // Convert to EC mS/cm
  Serial.printf("EC: %.2f mS/cm  TDS: %.0f PPM\n", r.ecMS, r.ppm);

  // 5. pH sensor (averaged 10 readings for stability)
  long adcSum = 0;
  for (int i = 0; i < 10; i++) {
    adcSum += analogRead(PIN_PH_ANALOG);
    delay(5);
  }
  float adcAvg     = adcSum / 10.0;
  float voltage    = adcAvg * (3.3 / 4095.0); // 12-bit ADC, 3.3V ref
  r.pH = PH_SLOPE * voltage + PH_INTERCEPT;   // Linear from calibration
  Serial.printf("pH: %.2f (voltage: %.3fV)\n", r.pH, voltage);

  // 6. A02YYUW — water level (ultrasonic, UART2)
  r.waterLevelPct = readUltrasonicLevel();
  Serial.printf("Water level: %d%%\n", r.waterLevelPct);

  return r;
}

// A02YYUW UART parser — returns distance in cm
int readUltrasonicLevel() {
  while (Serial2.available()) Serial2.read(); // flush stale data
  unsigned long t = millis();
  while (Serial2.available() < 4) {
    if (millis() - t > 500) return -1; // timeout
    yield();
  }
  uint8_t buf[4];
  Serial2.readBytes(buf, 4);
  if (buf[0] != 0xFF) return -1; // check header
  uint8_t checksum = (buf[0] + buf[1] + buf[2]) & 0xFF;
  if (checksum != buf[3]) return -1; // checksum fail
  int distanceMM = (buf[1] << 8) | buf[2];
  float distanceCM = distanceMM / 10.0;
  // Convert distance to level percentage
  float levelCM  = TANK_HEIGHT_CM - distanceCM;
  int   levelPct = (int)((levelCM / TANK_HEIGHT_CM) * 100.0);
  return constrain(levelPct, 0, 100);
}

// ─────────────────────────────────────────────────────
//  LED LIGHT AUTO-CONTROL (sensor-driven)
// ─────────────────────────────────────────────────────
void controlLED(int currentHour, float luxReading, bool preHarvestMode) {
  int offHour = preHarvestMode ? LED_PREHARVEST_OFF : LED_OFF_HOUR;
  bool inSchedule = (currentHour >= LED_ON_HOUR && currentHour < offHour);

  // Turn LED OFF if it's dark hours (plants need rest)
  if (!inSchedule) {
    digitalWrite(PIN_LED_RELAY, HIGH); // OFF
    return;
  }

  // During scheduled hours: auto-control based on lux
  if (luxReading > 25000) {
    // Too bright (direct sunlight) — LED not needed
    digitalWrite(PIN_LED_RELAY, HIGH); // OFF
  } else if (luxReading > 8000) {
    // Good light — LED ON for boost
    digitalWrite(PIN_LED_RELAY, LOW);  // ON
  } else {
    // Too dim — LED ON always during schedule
    digitalWrite(PIN_LED_RELAY, LOW);  // ON
  }
  Serial.printf("LED: %s (lux=%.0f hour=%d)\n",
    digitalRead(PIN_LED_RELAY) == LOW ? "ON" : "OFF",
    luxReading, currentHour);
}

// ─────────────────────────────────────────────────────
//  SAFETY CHECKS — run after every sensor read
// ─────────────────────────────────────────────────────
void runSafetyChecks(const SensorReadings& r) {
  // Water level too low — stop pump
  if (r.waterLevelPct < 10) {
    digitalWrite(PIN_PUMP_RELAY, HIGH); // STOP PUMP
    Serial.println("SAFETY: Pump stopped — water level critical");
  }
  // Water too hot — alert (in real code: send API alert)
  if (r.waterTempC > 26.0) {
    Serial.printf("ALERT: Water temp %.1f°C — add cooling!\n", r.waterTempC);
  }
  // Humidity too high — risk of fungal disease
  if (r.humidity > 75.0) {
    Serial.printf("ALERT: Humidity %.1f%% — improve ventilation!\n", r.humidity);
  }
  // pH out of range for lettuce
  if (r.pH < 5.0 || r.pH > 7.0) {
    Serial.printf("ALERT: pH %.2f out of range (5.0-7.0)!\n", r.pH);
  }
  // EC too high for lettuce
  if (r.ecMS > 1.4) {
    Serial.printf("ALERT: EC %.2f mS/cm too high for lettuce!\n", r.ecMS);
  }
}
